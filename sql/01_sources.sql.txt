-- ========================================
-- DATA FC – Nettoyage des données
-- Objectif : renommer les colonnes de la table ‘Instagram_followers_club_raw’
-- ========================================

 SELECT 
      string_field_0 AS Ligue,
      string_field_1 AS Club,
      string_field_2 AS Division,
      string_field_3 AS Instagram,
      string_field_4 AS Followers
  FROM `project_id.dataset_id.Instagram_followers_club_raw` 

-- ========================================
-- DATA FC – Nettoyage des données
-- Objectif : renommer ‘Angleterre’ par ‘Royaume-Uni’
-- ========================================

SELECT 
      *,
  	CASE 
      WHEN Pays = "Angleterre" THEN "Royaume-Uni" 
  		ELSE Pays 
  	END AS pays_clean
FROM `project_id.dataset_id.Club_geoloc_country_ladders`
