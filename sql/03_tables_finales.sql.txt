# subquery pour transformer capacite et moyenne en format string (pour pouvoir utiliser REPLACE ensuite), on ajoute les zéros manquants pour pouvoir transformer ensuite en INT

WITH aff_22_23_modified AS(
  SELECT Stade____Club,
    IF (Capacite = 400.0,0.4*1000,Capacite*1000) AS Capacite,
    CASE
      WHEN Spectateurs LIKE '%.__' THEN CONCAT(Spectateurs,'0')
      WHEN Spectateurs LIKE '%._' THEN CONCAT(Spectateurs,'00')
      WHEN Spectateurs LIKE '___' THEN CONCAT(Spectateurs,'000')
      WHEN Spectateurs LIKE '__' THEN CONCAT(Spectateurs,'000')
      WHEN Spectateurs LIKE '_' THEN CONCAT(Spectateurs,'000')
      ELSE Spectateurs
    END AS Spectateurs,
    CASE
      WHEN CAST(Moyenne AS STRING) LIKE '___' THEN CONCAT(CAST(Moyenne AS STRING),'')
      ELSE CAST(Moyenne*1000 AS STRING)
    END AS Moyenne,
    Matchs,
    A_guichets_fermes
  FROM ` project_id.dataset_id.transfermarket_data.Affluence clubs 22-23`
)
SELECT #dans la colonne Stade____Club, on sépare le nom du club du nom du stade
-- Column 1: Get the value after the ' <> ' delimiter (Index 1)
-- This is the easiest way to get the team name 'Olympique de Marseille'
  SPLIT(Stade____Club, ' <> ')[SAFE_OFFSET(1)] AS Club,
-- Column 2: Get the value BEFORE the ' <> ' delimiter (Index 0),
-- then remove the team name from that
  REPLACE(
    SPLIT(Stade____Club, ' <> ')[SAFE_OFFSET(0)],
    SPLIT(Stade____Club, ' <> ')[SAFE_OFFSET(1)], -- The team name 
    '') AS Stade,
    #on transforme Capacité, spectateurs et moyenne en INT tout en supprimant les '.'
  Capacite,
  CAST(REPLACE(Spectateurs,'.','') AS FLOAT64) AS Nombre_total_spectateurs_saison,
  CAST (Moyenne AS FLOAT64) AS Moyenne_affluence_match,
  Matchs AS Nombre_matchs_domicile,
  A_guichets_fermes,
  ROUND(SAFE_DIVIDE(CAST (Moyenne AS FLOAT64),Capacite),2) AS Pourcentage_remplissage
FROM aff_22_23_modified

SELECT
  ANY_VALUE(pays) AS pays,
  ANY_VALUE(championnat) AS championnat,
  Club,
  SUM(total_games) AS total_games,
  SUM(total_wins) AS total_wins,
  AVG(win_rate) AS avg_win_rate
FROM
  ` project_id.dataset_id.Sportif_win_rate_par_saison` AS win_rate_saison
GROUP BY
  Club;
