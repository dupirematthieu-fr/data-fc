WITH tab1 AS (
  SELECT  
      current_club_name,
      last_season,
# Agrégation par club pour avoir la valeur moyenne et max de market_value par joueur 
      ROUND(AVG(market_value_in_eur), 2) AS avg_market_value,
      MAX(highest_market_value_in_eur) AS highest_market_value
  FROM ` project_id.dataset_id.players`
# Filtre sur les deux dernières saisons
  WHERE last_season IN (2023, 2024)
  GROUP BY current_club_name, last_season
),

tab2 AS (
  SELECT
    *
  FROM tab1
# Première fonction pivot pour affichage de la première agrégation en colonne
  PIVOT(
    AVG(avg_market_value) FOR last_season IN (2023, 2024)
  )
),

tab3 AS (
  SELECT *
  FROM tab1
# Deuxième fonction pivot pour affichage de la deuxième agrégation en colonne
  PIVOT(
    MAX(highest_market_value) FOR last_season IN (2023, 2024)
  )
)

# Fusion des deux pivots
SELECT 
  tab3.current_club_name AS club,
  AVG(tab2._2023) AS avg_market_value_2023,
  AVG(tab2._2024) AS avg_market_value_2024,
  MAX(tab3._2023) AS highest_market_value_2023,
  MAX(tab3._2024) AS highest_market_value_2024
FROM tab2
INNER JOIN tab3
    USING (current_club_name)
GROUP BY club
ORDER BY club;

      SAFE_CAST(t1.Level AS INT64) AS Level_25_26,
      -- Conversion de 'm' en nombre (FLOAT64)
      SAFE_CAST(REPLACE(REPLACE(LOWER(t1.Total_market_value_2025_2026), '€', ''), 'm', '') AS FLOAT64) * 1000000.0 AS Total_Value_25_26
    FROM 
      ` project_id.dataset_id.Evolution_valeur_marchande_25_26` t1
    WHERE 
      t1.Total_market_value_2025_2026 IS NOT NULL AND t1.___age IS NOT NULL
  ),

  -- 2. Nettoyage(24-25)
  Data_24_25 AS (
    SELECT
      t2.Club,
      -- Conversion de 'm''' en nombre 
      SAFE_CAST(REPLACE(REPLACE(LOWER(t2.Total_market_value_2024_2025), '€', ''), 'm', '') AS FLOAT64) * 1000000.0 AS Total_Value_24_25
    FROM 
      ` project_id.dataset_id.Evolution_valeur_marchande_24_25` t2
    WHERE 
      t2.Total_market_value_2024_2025 IS NOT NULL
  )

-- 3. Jointure et Calcul du Potentiel Pondéré
SELECT
  d25.Club,
  d25.Pays,
  d25.Division,
  d25.Age_Moyen_25_26,
  CAST(d25.Total_Value_25_26 AS INT64) AS Total_Value_25_26_Unites, 
  
  -- Taux de croissance 
  SAFE_DIVIDE((d25.Total_Value_25_26 - d24.Total_Value_24_25), d25.Total_Value_25_26) AS Progression_Taux_Club,

  -- SCORE FINAL : Taux de Croissance * Facteur de Jeunesse
  (
    SAFE_DIVIDE((d25.Total_Value_25_26 - d24.Total_Value_24_25), d24.Total_Value_24_25) * (CASE
        -- Haut Potentiel : Âge moyen < 23 ans (facteur 2.0x)
        WHEN d25.Age_Moyen_25_26 < 23 THEN 2.0
        -- Jeunes : Âge moyen entre 23 et 25 ans (facteur 1.5x)
        WHEN d25.Age_Moyen_25_26 >= 23 AND d25.Age_Moyen_25_26 < 25 THEN 1.5
        -- Moins Intéressant : Âge moyen >= 25 ans (facteur 1.0x)
        ELSE 1.0
    END)
  ) AS Score_Potentiel
  
FROM
  Data_25_26 d25
INNER JOIN
  Data_24_25 d24 ON d25.Club = d24.Club

WHERE
  d25.Total_Value_25_26 IS NOT NULL 
  AND d24.Total_Value_24_25 > 0 
  
  
ORDER BY
  Progression_Taux_Club DESC
;

#average minute per player passé dans le centre de formation et indice de formation
SELECT
  Club AS club_name,
 SUM(CAST(REPLACE(CAST(Minutes_Jouees AS STRING), ' ', '') AS INT64)) AS sum_minutes,
  SUM(CAST(REPLACE(CAST(Nombre_Joueurs AS STRING), ' ', '') AS INT64)) AS sum_players
  , SUM(CAST(REPLACE(CAST(Minutes_Jouees AS STRING), ' ', '') AS INT64)) /
  SUM(CAST(REPLACE(CAST(Nombre_Joueurs AS STRING), ' ', '') AS INT64)) AS avg_minutes_per_player
  ,Indice_Formation AS indice_formation
FROM ` project_id.dataset_id.Centre_de_formation_raw`
GROUP BY Club, Indice_Formation
ORDER BY avg_minutes_per_player DESC;

# taux de win 

SELECT
  cg.club_id,
  season,
  COUNT(*) AS total_games, 
  SUM(is_win) AS total_wins,      
  ROUND(SUM(is_win) / COUNT(*), 2) AS win_rate 
FROM ` project_id.dataset_id.club_games` AS cg
JOIN ` project_id.dataset_id.games` AS g
  ON g.game_id = cg.game_id
GROUP BY cg.club_id, season;
